Experiment 4: Hamming Code Generation, Error Detection, and CorrectionAim: Write a program to generate Hamming Code for error detection and error correction.   C#include <stdio.h>
#include <math.h>

int input[32];
int code[32];

int ham_calc(int position, int total_len);

int main() {
    int n, p_n = 0, c_l, i, j, k;
    int error_pos = 0;

    printf("Please enter the length of the Data word: ");
    scanf("%d", &n);

    printf("Please enter the Data word (bit by bit):\n");
    for (i = 0; i < n; i++) {
        scanf("%d", &input[i]);
    }

    // Calculate number of parity bits required: 2^p_n >= n + p_n + 1
    while ((int)pow(2, p_n) < (n + p_n + 1)) {
        p_n++;
    }

    c_l = p_n + n;
    j = 0;

    // Initialize code word (mark parity positions with 0)
    for (i = 1; i <= c_l; i++) {
        if (i == (int)pow(2, k)) {
            code[i - 1] = 0;
            k++;
        } else {
            code[i - 1] = input[j];
            j++;
        }
    }

    // Calculate parity bits
    for (i = 0; i < p_n; i++) {
        int position = (int)pow(2, i);
        int value = ham_calc(position, c_l);
        code[position - 1] = value;
    }

    printf("\nThe Calculated Code word is: ");
    for (i = 0; i < c_l; i++) {
        printf("%d ", code[i]);
    }
    printf("\n");

    printf("Please enter the received Code word:\n");
    for (i = 0; i < c_l; i++) {
        scanf("%d", &code[i]);
    }

    // Check for errors at the receiver side
    for (i = 0; i < p_n; i++) {
        int position = (int)pow(2, i);
        int value = ham_calc(position, c_l);
        if (value != 0) {
            error_pos += position;
        }
    }

    if (error_pos == 0) {
        printf("The received Code word is Correct\n");
    } else {
        printf("Error at bit position: %d\n", error_pos);
    }

    return 0;
}

int ham_calc(int position, int total_len) {
    int count = 0, i, j;
    i = position - 1;

    while (i < total_len) {
        for (j = i; j < i + position && j < total_len; j++) {
            if (code[j] == 1) {
                count++;
            }
        }
        i = i + 2 * position;
    }

    if (count % 2 == 0)
        return 0;
    else
        return 1;
}

int main() {
    int no_of_frames = 3, N = 2;
    int count = 0;
    int i = 1, j;

    srand(time(NULL));

    while (i <= no_of_frames) {
        int x = 0;

        // Transmit current window frames
        for (j = i; j < i + N && j <= no_of_frames; j++) {
            printf("\nSending frame %d", j);
            count++;
        }

        // Process ACKs
        for (j = i; j < i + N && j <= no_of_frames; j++) {
            int flag = rand() % 2;
            if (flag) {
                printf("\nAcknowledgement for window frame %d received", j);
                x++;
            } else {
                printf("\nCumulative acknowledgement not received whose first packet is %d", j);
                printf("\nRetransmitting the same window...\n");
                break;
            }
        }
        i += x;
    }

    printf("\n\nTotal number of transmissions: %d\n", count);
    return 0;
}

void dijkstra(int graph[V][V], int src) {
    int dist[V];
    bool sptSet[V];

    for (int i = 0; i < V; i++) {
        dist[i] = INT_MAX;
        sptSet[i] = false;
    }

    dist[src] = 0;

    for (int count = 0; count < V - 1; count++) {
        int u = minDistance(dist, sptSet);
        sptSet[u] = true;

        for (int v = 0; v < V; v++) {
            if (!sptSet[v] && graph[u][v] && dist[u] != INT_MAX
                && dist[u] + graph[u][v] < dist[v]) {
                dist[v] = dist[u] + graph[u][v];
            }
        }
    }

    printSolution(dist, V);
}

#define INF 999
#define N 6

void printTable(int dist[N][N], int nodes) {
    printf("\nRouting Table:\n");
    printf("Node | ");
    for (int i = 0; i < nodes; i++) {
        printf("%c\t", 'A' + i);
    }
    printf("\n--------------------------------------------------\n");

    for (int i = 0; i < nodes; i++) {
        printf("%c    | ", 'A' + i);
        for (int j = 0; j < nodes; j++) {
            if (dist[i][j] == INF)
                printf("INF\t");
            else
                printf("%d\t", dist[i][j]);
        }
        printf("\n");
    }
}

int main() {
    int graph[N][N] = {
        {0,   1,   INF, INF, INF, INF},
        {1,   0,   2,   INF, 1,   INF},
        {INF, 2,   0,   3,   INF, INF},
        {INF, INF, 3,   0,   2,   INF},
        {INF, 1,   INF, 2,   0,   3  },
        {INF, INF, INF, INF, 3,   0  }
    };

    int dist[N][N];
    int nodes = N;

    for (int i = 0; i < nodes; i++) {
        for (int j = 0; j < nodes; j++) {
            dist[i][j] = graph[i][j];
        }
    }

    // Dynamic programming (Distance Vector update)
    for (int k = 0; k < nodes; k++) {
        for (int i = 0; i < nodes; i++) {
            for (int j = 0; j < nodes; j++) {
                if (dist[i][k] + dist[k][j] < dist[i][j]) {
                    dist[i][j] = dist[i][k] + dist[k][j];
                }
            }
        }
    }

    printTable(dist, nodes);
    return 0;
}

#define INF INT_MAX

int minDistance(int dist[], int visited[]) {
    int min = INF, min_index = -1;

    for (int v = 0; v < N; v++) {
        if (visited[v] == 0 && dist[v] < min) {
            min = dist[v];
            min_index = v;
        }
    }
    return min_index;
}

void printBroadcastTree(int parent[], int graph[N][N]) {
    printf("Edge \tWeight\n");
    for (int i = 1; i < N; i++) {
        printf("%c - %c \t%d\n", 'A' + parent[i], 'A' + i, graph[i][parent[i]]);
    }

void broadcastTree(int graph[N][N], int src) {
    int dist[N];
    int parent[N];
    int visited[N];

    for (int i = 0; i < N; i++) {
        dist[i] = INF;
        visited[i] = 0;
    }

    dist[src] = 0;
    parent[src] = -1;

    for (int count = 0; count < N - 1; count++) {
        int u = minDistance(dist, visited);
        visited[u] = 1;

        for (int v = 0; v < N; v++) {
            if (graph[u][v] && visited[v] == 0 && graph[u][v] < dist[v]) {
                parent[v] = u;
                dist[v] = graph[u][v];
            }
        }
    }

    printBroadcastTree(parent, graph);
}
